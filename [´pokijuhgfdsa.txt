    <script>
        /* =========================================================
           LÓGICA INTERATIVA DO JOGO DE QUEBRA-CABEÇA
           ========================================================= */

        // Estado do Jogo
        let estadoAtual = [0, 1, 2, 3, 4, 5, 6, 7, 8];
        let pecaSelecionadaIndex = null;
        let movimentos = 0;
        let tempoSegundos = 0;
        let timerInterval = null;
        let jogoIniciado = false;
        let jogoConcluido = false;

        // Elementos DOM
        const puzzleGrid = document.getElementById('puzzleGrid');
        const txtMovimentos = document.getElementById('txtMovimentos');
        const txtTempo = document.getElementById('txtTempo');
        const vitoriaOverlay = document.getElementById('vitoriaOverlay');
        const vitMovimentos = document.getElementById('vitMovimentos');
        const vitTempo = document.getElementById('vitTempo');

        // SVG DataURI usado para preencher dinamicamente as peças do Quebra-cabeça
        const svgString = `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 300 300">
            <rect width="300" height="300" fill="#eaf4ff"/>
            <circle cx="150" cy="150" r="130" fill="#ffdcf0"/>
            <circle cx="150" cy="150" r="105" fill="#ffffff" opacity="0.6"/>
            <g transform="translate(15, 20) scale(1.4)">
                <path d="M75 25 L83 45 L95 30 L107 45 L115 25 L110 50 L80 50 Z" fill="#ffc735" stroke="#062752" stroke-width="3"/>
                <rect x="35" y="45" width="110" height="40" rx="20" fill="#209fe9" stroke="#062752" stroke-width="5" transform="rotate(-5 90 65)"/>
                <rect x="110" y="52" width="40" height="12" rx="4" fill="#7edbff" transform="rotate(2 130 58)"/>
                <rect x="35" y="70" width="110" height="90" rx="45" fill="#fff1ed" stroke="#062752" stroke-width="5"/>
                <ellipse cx="65" cy="105" rx="10" ry="16" fill="#062752"/>
                <circle cx="62" cy="98" r="3" fill="#ffffff"/>
                <ellipse cx="115" cy="105" rx="10" ry="16" fill="#062752"/>
                <circle cx="112" cy="98" r="3" fill="#ffffff"/>
                <ellipse cx="52" cy="125" rx="10" ry="5" fill="#ff91b5"/>
                <ellipse cx="128" cy="125" rx="10" ry="5" fill="#ff91b5"/>
                <path d="M 83 126 Q 90 136 97 126" fill="transparent" stroke="#f25291" stroke-width="4" stroke-linecap="round"/>
                <text x="145" y="40" font-size="20" fill="#ffc83d">★</text>
                <text x="20" y="160" font-size="16" fill="#ff6da7">♥</text>
            </g>
        </svg>`;

        const bgDataUrl = "data:image/svg+xml;charset=utf-8," + encodeURIComponent(svgString);

        // Inicializar a aplicação ao carregar
        window.addEventListener('DOMContentLoaded', () => {
            reiniciarJogo();
        });

        // Embaralha o vetor de posições
        function embaralhar(array) {
            let copia = [...array];
            let n = copia.length;
            // Garante que não fique igual ao gabarito após embaralhar
            do {
                for (let i = n - 1; i > 0; i--) {
                    const j = Math.floor(Math.random() * (i + 1));
                    [copia[i], copia[j]] = [copia[j], copia[i]];
                }
            } while (copia.every((val, idx) => val === idx));
            return copia;
        }

        // Renderiza o grid do Quebra-cabeça
        function renderizarGrid() {
            puzzleGrid.innerHTML = '';
            
            estadoAtual.forEach((valorPeca, posicaoIndex) => {
                const peca = document.createElement('div');
                peca.className = 'puzzle-piece';
                
                if (pecaSelecionadaIndex === posicaoIndex) {
                    peca.classList.add('selected');
                }

                // Cálculo das posições de fundo da imagem cortada (3x3)
                const originalRow = Math.floor(valorPeca / 3);
                const originalCol = valorPeca % 3;
                const bgX = originalCol * 50; // 0%, 50%, 100%
                const bgY = originalRow * 50;

                peca.style.backgroundImage = `url("${bgDataUrl}")`;
                peca.style.backgroundPosition = `${bgX}% ${bgY}%`;

                // Número pequeno de auxílio
                const numGuia = document.createElement('span');
                numGuia.className = 'indice-guia';
                numGuia.innerText = valorPeca + 1;
                peca.appendChild(numGuia);

                // Evento de Clique para Trocar Peças
                peca.addEventListener('click', () => clicarPeca(posicaoIndex));

                puzzleGrid.appendChild(peca);
            });
        }

        // Lógica de Seleção e Troca
        function clicarPeca(index) {
            if (jogoConcluido) return;

            // Iniciar Cronômetro no primeiro movimento
            if (!jogoIniciado) {
                iniciarTimer();
                jogoIniciado = true;
            }

            if (pecaSelecionadaIndex === null) {
                // Primeira peça selecionada
                pecaSelecionadaIndex = index;
            } else if (pecaSelecionadaIndex === index) {
                // Desmarcar se clicar na mesma
                pecaSelecionadaIndex = null;
            } else {
                // Segunda peça selecionada: Trocar de lugar
                const temp = estadoAtual[pecaSelecionadaIndex];
                estadoAtual[pecaSelecionadaIndex] = estadoAtual[index];
                estadoAtual[index] = temp;

                pecaSelecionadaIndex = null;
                movimentos++;
                txtMovimentos.innerText = movimentos;

                // Checa condição de vitória
                checarVitoria();
            }

            renderizarGrid();
        }

        // Inicia o Timer de Tempo
        function iniciarTimer() {
            clearInterval(timerInterval);
            timerInterval = setInterval(() => {
                tempoSegundos++;
                const mins = String(Math.floor(tempoSegundos / 60)).padStart(2, '0');
                const segs = String(tempoSegundos % 60).padStart(2, '0');
                txtTempo.innerText = `${mins}:${segs}`;
            }, 1000);
        }

        // Reseta tudo e embaralha o jogo
        function reiniciarJogo() {
            clearInterval(timerInterval);
            tempoSegundos = 0;
            movimentos = 0;
            jogoIniciado = false;
            jogoConcluido = false;
            pecaSelecionadaIndex = null;

            txtMovimentos.innerText = '0';
            txtTempo.innerText = '00:00';
            vitoriaOverlay.style.display = 'none';

            estadoAtual = embaralhar([0, 1, 2, 3, 4, 5, 6, 7, 8]);
            renderizarGrid();
        }

        // Verifica se todas as peças estão na posição correta
        function checarVitoria() {
            const venceu = estadoAtual.every((val, idx) => val === idx);

            if (venceu) {
                jogoConcluido = true;
                clearInterval(timerInterval);

                vitMovimentos.innerText = movimentos;
                vitTempo.innerText = txtTempo.innerText;

                // Ativar efeito de Confetes
                if (typeof confetti === 'function') {
                    confetti({
                        particleCount: 100,
                        spread: 70,
                        origin: { y: 0.6 }
                    });
                }

                setTimeout(() => {
                    vitoriaOverlay.style.display = 'flex';
                }, 300);
            }
        }

        // Drawer de Menu Mobile
        function abrirMenu() {
            document.getElementById("menuLateral").classList.add("aberto");
        }

        function fecharMenu() {
            document.getElementById("menuLateral").classList.remove("aberto");
        }
